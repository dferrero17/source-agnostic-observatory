% The autonomic wrapper manager: three self-* loops over one generated artifact.
% MAPE-K framing (Monitor-Analyze-Plan-Execute over shared Knowledge).
\begin{tikzpicture}[
  font=\scriptsize,
  stage/.style={draw, rounded corners=1pt, align=center, minimum height=16pt,
                inner sep=3pt, fill=gray!8},
  art/.style={draw, align=center, minimum height=16pt, inner sep=3pt,
              fill=gray!20},
  loop/.style={-{Stealth[length=4pt]}, thick},
  node distance=16pt and 22pt]

% the shared artifact at the centre of all three loops
\node[art] (wrapper) {wrapper artifact\\(generated code +\\watermarks + status)};

% self-generating (left)
\node[stage, left=34pt of wrapper, align=center] (gen)
  {\textbf{self-generating}\\metadata + sample\\$\rightarrow$ template fill\\(LLM, tool-probed)};
% self-deploying (top)
\node[stage, above=20pt of wrapper, align=center] (dep)
  {\textbf{self-deploying}\\spawn, supervise,\\checkpoint-resume};
% self-healing (right)
\node[stage, right=34pt of wrapper, align=center] (heal)
  {\textbf{self-healing}\\crash + trace\\$\rightarrow$ LLM repair\\($\le 3$ attempts)};

\draw[loop] (gen.east) -- (wrapper.west);
\draw[loop] (wrapper.north) -- (dep.south);
\draw[loop] (dep.east) to[out=0, in=90] (heal.north);
\draw[loop] (heal.south) to[out=-90, in=0] (wrapper.east);
\draw[loop] (wrapper.east) to[out=20, in=200] node[below, pos=.5]{} (heal.west);
\draw[loop, dashed] (heal.west) to[out=160, in=20]
  node[above, pos=.5]{redeploy} (dep.east);
\end{tikzpicture}
